122 Double_t Theta(
123 const Ex03dParticle& p) // Returns the angle between momenta of particles
124 {
125 Double_t v = P() * p.P();
126 if (v == 0)
127 v = 1;
128 else
129 v = (fPx * p.Px() + fPy * p.Py() + fPz * p.Pz()) / v;
130 if (v > 1)
131 v = 1;
132 else if (v < -1)
133 v = -1; // just a precaution
134 return TMath::ACos(v);
135 }
