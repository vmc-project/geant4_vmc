138void TG4Limits::Initialize(

140{
142
143 fMaxTime = cuts[kTOFMAX];

226// The limits objects are attached to logical volumes and so shared by all
227// threads in multithreaded mode; the max step set on fly is kept per thread
228G4ThreadLocal const TG4Limits* gLimitsModifiedOnFly = nullptr;
229G4ThreadLocal G4double gMaxStepOnFly = DBL_MAX;
230} // namespace
231
232//_____________________________________________________________________________
233void TG4Limits::SetCurrentMaxAllowedStep(G4double step)
234{
236
237 // G4cout << "TG4Limits::SetCurrentMaxStep: in " << GetName()
238 // << " step " << step << G4endl;
239
240 gLimitsModifiedOnFly = this;
241 gMaxStepOnFly = step;
242}
243
244//_____________________________________________________________________________
245void TG4Limits::SetMaxAllowedStepBack()
246{
249
250 if (gLimitsModifiedOnFly == this) {
251 gLimitsModifiedOnFly = nullptr;
252 }
253}

255//_____________________________________________________________________________
256G4double TG4Limits::GetMaxUserStep() const
257{

262}

272//_____________________________________________________________________________
273void TG4Limits::Print() const
274{
276
277 G4cout << "\"" << fName << "\" limits:" << G4endl;
278 G4cout << " Max step length (mm): " << GetMaxUserStep() / mm << G4endl;
279 G4cout << " Max track length (mm): " << fMaxTrack / mm << G4endl;
280 G4cout << " Max time (s) " << fMaxTime / s << G4endl;
281 G4cout << " Min kin. energy (MeV) " << fMinEkine / MeV << G4endl;
282 G4cout << " Min range (mm): " << fMinRange / mm << G4endl;
283
284 if (!fIsCut) G4cout << " No special cuts. " << G4endl;
285 if (!fIsControl) G4cout << " No special controls. " << G4endl;
286
287 if (fIsCut) fCutVector.Print();
288 if (fIsControl) fControlVector.Print();
289}
